```mermaid

    U["User"] --> UI["Web Client"]
    UI --> API["REST API"]
    API --> AUTH["JWT Authentication"]
    API --> UP["Secure File Upload"]
    API --> AN["Resume Analysis"]
    AN --> OL["Ollama LLM"]
    AN --> F1["ATS Scoring"]
    AN --> F2["Job Matching"]
    AN --> F3["Interview Prep"]
    AN --> F4["Cover-Letter Generation"]
    API --> ORM["Prisma"]

    DB --> DASH["Analytics Dashboard"]
    DASH --> UI
```
